name: Build Windows

on:
  push:
    branches: [main]
  workflow_dispatch:
  release:
    types: [published]

permissions:
  contents: write

concurrency:
  group: windows-${{ github.event.release.tag_name || github.ref }}
  cancel-in-progress: false

jobs:
  build:
    runs-on: windows-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: '24'
          cache: npm

      - name: Verify release version
        if: github.event_name == 'release'
        shell: pwsh
        env:
          RELEASE_TAG: ${{ github.event.release.tag_name }}
        run: |
          $version = (Get-Content package.json -Raw | ConvertFrom-Json).version
          if ($env:RELEASE_TAG -ne "v$version") { throw "Release tag must be v$version" }

      - name: Install dependencies
        run: npm ci

      - name: Syntax check
        run: npm run check

      - name: Regression tests
        run: npm test

      - name: Build installer
        run: npm run dist -- --publish never
        env:
          CSC_IDENTITY_AUTO_DISCOVERY: 'false'

      - name: Upload installer artifact
        uses: actions/upload-artifact@v4
        with:
          name: Fishing-Chat-Game-Windows
          path: dist/*.exe
          if-no-files-found: error

      - name: Attach installer to published release
        if: github.event_name == 'release'
        shell: pwsh
        env:
          GH_TOKEN: ${{ github.token }}
          GH_REPO: ${{ github.repository }}
          RELEASE_TAG: ${{ github.event.release.tag_name }}
        run: |
          $installers = @(Get-ChildItem -LiteralPath dist -Filter '*.exe' -File)
          if ($installers.Count -eq 0) { throw 'Installer not found' }
          foreach ($installer in $installers) {
            gh release upload $env:RELEASE_TAG $installer.FullName --repo $env:GH_REPO --clobber
            if ($LASTEXITCODE -ne 0) { throw 'Release upload failed' }
          }
